cmake_minimum_required(VERSION 3.25)
project(AYAbility VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)

add_library(AYAbilityCore STATIC src/Ability.cpp src/Runtime.cpp src/Derived.cpp src/Combo.cpp)
add_library(AYAbility::Core ALIAS AYAbilityCore)
target_compile_features(AYAbilityCore PUBLIC cxx_std_23)
target_include_directories(AYAbilityCore PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>)
if(MSVC)
    target_compile_options(AYAbilityCore PRIVATE /W4 /utf-8 /Zc:__cplusplus)
endif()
find_package(nlohmann_json CONFIG REQUIRED)
add_library(AYAbilityAuthoring STATIC src/Authoring.cpp)
add_library(AYAbility::Authoring ALIAS AYAbilityAuthoring)
target_link_libraries(AYAbilityAuthoring PUBLIC AYAbilityCore nlohmann_json::nlohmann_json)
if(MSVC)
    target_compile_options(AYAbilityAuthoring PRIVATE /utf-8 /Zc:__cplusplus)
endif()

option(AY_ABILITY_BUILD_ENTITY "Build World/Stats/Combat ability integration (engine build)" OFF)
option(AY_ABILITY_BUILD_PHYSICS2D "Build 2D physics delivery adapter (engine build)" ON)
option(AY_ABILITY_BUILD_PHYSICS3D "Build 3D physics delivery adapter (engine build)" ON)
if(PROJECT_IS_TOP_LEVEL AND AY_ABILITY_BUILD_ENTITY)
    message(FATAL_ERROR "AY_ABILITY_BUILD_ENTITY requires the root engine build")
endif()
set(AY_ABILITY_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}" CACHE INTERNAL "Ability integration sources")
function(ay_ability_add_entity_integration)
    if(NOT AY_ABILITY_BUILD_ENTITY)
        return()
    endif()
    if(NOT TARGET AYEntityCombatIntegration)
        message(FATAL_ERROR "AY_ABILITY_BUILD_ENTITY requires AYCombat::Entity")
    endif()
    add_library(AYEntityAbilityIntegration STATIC "${AY_ABILITY_SOURCE_DIR}/src/EntityIntegration.cpp"
        "${AY_ABILITY_SOURCE_DIR}/src/DeliveryIntegration.cpp")
    add_library(AYAbility::Entity ALIAS AYEntityAbilityIntegration)
    target_link_libraries(AYEntityAbilityIntegration PUBLIC AYAbilityAuthoring AYEntityCombatIntegration)
    if(MSVC)
        target_compile_options(AYEntityAbilityIntegration PRIVATE /utf-8 /Zc:preprocessor /Zc:__cplusplus)
    endif()
    if(BUILD_TESTING)
        add_executable(AYAbility_EntityTests "${AY_ABILITY_SOURCE_DIR}/unittest/main.cpp"
            "${AY_ABILITY_SOURCE_DIR}/unittest/Test_Entity.cpp")
        target_link_libraries(AYAbility_EntityTests PRIVATE AYEntityAbilityIntegration AYTest)
        if(MSVC)
            target_compile_options(AYAbility_EntityTests PRIVATE /utf-8 /Zc:preprocessor /Zc:__cplusplus)
        endif()
        ay_add_test(AYAbility_EntityTests CASE_REPORT LABELS "ability;ability-integration;ability-full;integration" TIMEOUT 60 RUN_SERIAL)
    endif()
    add_executable(AYAbility_GameplayExample "${AY_ABILITY_SOURCE_DIR}/examples/Gameplay.cpp")
    target_link_libraries(AYAbility_GameplayExample PRIVATE AYEntityAbilityIntegration)
    if(MSVC)
        target_compile_options(AYAbility_GameplayExample PRIVATE /utf-8 /Zc:preprocessor /Zc:__cplusplus)
    endif()
endfunction()

function(ay_ability_add_physics2d_integration)
    if(NOT AY_ABILITY_BUILD_ENTITY OR NOT AY_ABILITY_BUILD_PHYSICS2D)
        return()
    endif()
    if(NOT TARGET AYEntityAbilityIntegration OR NOT TARGET AYCombatPhysics2D)
        message(FATAL_ERROR "AYAbility 2D delivery requires Entity and Combat 2D integrations")
    endif()
    add_library(AYAbilityPhysics2D STATIC "${AY_ABILITY_SOURCE_DIR}/src/Physics2DIntegration.cpp")
    add_library(AYAbility::Physics2D ALIAS AYAbilityPhysics2D)
    target_link_libraries(AYAbilityPhysics2D PUBLIC AYEntityAbilityIntegration
        AYCombatPhysics2D)
    if(MSVC)
        target_compile_options(AYAbilityPhysics2D PRIVATE /utf-8 /Zc:preprocessor /Zc:__cplusplus)
    endif()
    if(BUILD_TESTING)
        add_executable(AYAbility_Physics2DTests "${AY_ABILITY_SOURCE_DIR}/unittest/main.cpp"
            "${AY_ABILITY_SOURCE_DIR}/unittest/Test_Physics2D.cpp")
        target_link_libraries(AYAbility_Physics2DTests PRIVATE AYAbilityPhysics2D AYTest)
        target_compile_definitions(AYAbility_Physics2DTests PRIVATE
            AY_ABILITY_EXAMPLE_DIR="${AY_ABILITY_SOURCE_DIR}/examples")
        if(MSVC)
            target_compile_options(AYAbility_Physics2DTests PRIVATE /utf-8 /Zc:preprocessor /Zc:__cplusplus)
        endif()
        ay_add_test(AYAbility_Physics2DTests CASE_REPORT LABELS "ability;ability-physics2d;integration" TIMEOUT 60 RUN_SERIAL)
    endif()
endfunction()

if(BUILD_TESTING)
    if(NOT TARGET AYTest)
        add_subdirectory(${CMAKE_CURRENT_SOURCE_DIR}/../../AYFoundation/AYTest ${CMAKE_CURRENT_BINARY_DIR}/foundation/test)
        add_subdirectory(${CMAKE_CURRENT_SOURCE_DIR}/../../AYFoundation/AYPlatform ${CMAKE_CURRENT_BINARY_DIR}/foundation/platform)

    ay_add_test_tiers(AYAbility_UnitTests PREFIX ability TIMEOUT 60)
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/unittest/test-features.json" "{\"entity\":\"${AY_ABILITY_BUILD_ENTITY}\"}\n")
endif()
